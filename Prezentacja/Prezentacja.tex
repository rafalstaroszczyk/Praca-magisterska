% Define \smartfootcite to merge identical citations
\makeatletter
\newcommand{\smartfootcite}[2][]{%
	\begingroup
	\edef\tempkey{#2}% Store citation key
	\edef\temppages{#1}% Store page number(s)
	\edef\tempid{\tempkey:\temppages}% Unique identifier for key-page combination
	\ifcsname smartfoot@\tempid\endcsname
	% If the citation exists, reuse the footnote number
	\footnotemark[\csname smartfoot@\tempid\endcsname]%
	\else
	% Otherwise, create a new footnote and store its number
	%\footfullcite[#1]{#2}%
	\footnote[frame]{\fullcite[#1]{#2}}%
	\edef\tempnum{\number\value{footnote}}%
	\expandafter\xdef\csname smartfoot@\tempid\endcsname{\tempnum}%
	\fi
	\endgroup
}
\makeatother
