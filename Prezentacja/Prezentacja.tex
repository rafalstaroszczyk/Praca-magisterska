\section{Warunki endoodwracalne}
\begin{frame}{Warunki endoodwracalne}
\begin{columns}
	\begin{column}{0.5\textwidth}
		\begin{block}{}
			\begin{itemize}
				\item Równowaga wewnętrzna
				\item Nierównowaga z otoczeniem
			\end{itemize}
			\begin{gather*}
				x = T_{h} - T_{hw}, \\
				y = T_{cw} - T_{c}, \\
				Q_h \propto x, \\
				Q_c \propto y
			\end{gather*}
		\end{block}
	\end{column}
	\begin{column}{0.5\textwidth}
		\begin{figure}[H]
			\centering
			\includegraphics[width=\textwidth]{schemat_endo_carnot.pdf}

\end{frame}

\begin{frame}{Endoodwracalny silnik Carnota}
\begin{columns}
	\begin{column}{0.5\textwidth}
		\begin{block}{Przemiany izotermiczne}
			\begin{gather*}
				Q_{1-2} = \alpha_c \tau_c \pab{T_c - T_{cw}} = - \alpha_c \tau_c y, \\
				Q_{3-4} = \alpha_h \tau_h \pab{T_h - T_{hw}} =   \alpha_h \tau_h x.
			\end{gather*}
			\begin{gather*}
				\frac{Q_{1-2}}{T_{cw}} = -\frac{Q_{3-4}}{T_{hw}}, \\
				\frac{\tau_h}{\tau_c} = \frac{\alpha_c y \pab{T_h - x}}{\alpha_h x \pab{T_c + y}}
			\end{gather*}
		\end{block}
	\end{column}
	\begin{column}{0.5\textwidth}
		\begin{figure}[H]
			\centering
			\includegraphics[width=\textwidth]{pv_silnik_endo_carnot_70.pdf}
			\caption*{Schemat silnika Carnota}
		\end{figure}
	\end{column}
\end{columns}
\end{frame}

\begin{frame}{Endoodwracalny silnik Carnota}
\begin{block}{Moc silnika}
	\begin{equation*}
		P\pab{x,y} = \frac{\alpha_h \alpha_c xy}{\zeta} \frac{T_h-T_c-x-y}{\alpha_h T_c x + \alpha_c T_h y + \pab{\alpha_h-\alpha_c}xy}
	\end{equation*}
\end{block}
\begin{block}{Warunki maksimum}
	\begin{align*}
		\pdv{P}{x} &= 0, &
		\pdv{P}{y} &= 0, \\
		\pdv[order=2]{P}{x} &< 0, &
		\pdv[order=2]{P}{x}\pdv[order=2]{P}{y} - \pdv{P}{x,y}\pdv{P}{y,x} &> 0
	\end{align*}
\end{block}
\end{frame}

\begin{frame}{Endoodwracalny silnik Carnota}
\begin{block}{Rozwiązanie}
	\begin{align*}
		\frac{x_0}{T_h} &= \frac{1 - \sqrt{\frac{T_c}{T_h}}}{1 + \sqrt{\frac{\alpha_h}{\alpha_c}}}, &
		\frac{y_0}{T_c} &= \frac{\sqrt{\frac{T_h}{T_c}} - 1}{1 + \sqrt{\frac{\alpha_c}{\alpha_h}}}
	\end{align*}
\end{block}
\begin{block}{Sprawność i moc w punkcie maksymalnej mocy}
	\begin{gather*}
		\eta = 1 - \frac{T_{cw}}{T_{hw}} = 1 - \sqrt{\frac{T_c}{T_h}}, \\
		P = \frac{\alpha_h \alpha_c}{\zeta}\pab{\frac{\sqrt{T_h}-\sqrt{T_c}}{\sqrt{\alpha_h}+\sqrt{\alpha_c}}}^2
	\end{gather*}
\end{block}
\end{frame}

\begin{frame}{Endoodwracalny silnik Otto}
\begin{columns}
	\begin{column}{0.5\textwidth}
		\begin{block}{Zmiana temperatury}
			\begin{gather*}
				\odv{T}{t} = -\alpha_h \pab{T-T_h}, \\
				T\pab{t=0} = T_2, \\
				T\pab{t=\tau_h} = T_3
			\end{gather*}
			\begin{gather*}
				\odv{T}{t} = -\alpha_c \pab{T-T_c}, \\
				T\pab{t=0} = T_4, \\
				T\pab{t=\tau_h} = T_1
			\end{gather*}
			\begin{equation*}
				\frac{T_1}{T_2} = \frac{T_4}{T_3} = \kappa
			\end{equation*}
		\end{block}
	\end{column}
	\begin{column}{0.5\textwidth}
		\begin{figure}[H]
			\centering
			\includegraphics[width=\textwidth]{pv_silnik_endo_otto_70.pdf}

		\end{figure}
	\end{column}
\end{columns}
\end{frame}

\begin{frame}{Endoodwracalny silnik Otto}
\begin{block}{Temperatury}
	\begin{align*}
		T_1 &= \frac{
			T_c e^{\alpha_h \tau_h} \pab{e^{\alpha_c \tau_c} - 1} + \kappa T_h \pab{e^{\alpha_h \tau_h} - 1}
		}{
			e^{\alpha_h \tau_h + \alpha_c \tau_c} - 1
		}, \\
		T_2 &= \frac{
			T_c e^{\alpha_h \tau_h} \pab{e^{\alpha_c \tau_c} - 1} + \kappa T_h \pab{e^{\alpha_h \tau_h} - 1}
		}{
			\kappa \pab{e^{\alpha_h \tau_h + \alpha_c \tau_c} - 1}
		}, \\
		T_3 &= \frac{
			T_c \pab{e^{\alpha_c \tau_c} - 1} + \kappa T_h e^{\alpha_c \tau_c} \pab{e^{\alpha_h \tau_h} - 1}
		}{
			\kappa \pab{e^{\alpha_h \tau_h + \alpha_c \tau_c} - 1}
		}, \\
		T_4 &= \frac{
			T_c \pab{e^{\alpha_c \tau_c} - 1} + \kappa T_h e^{\alpha_c \tau_c} \pab{e^{\alpha_h \tau_h} - 1}
		}{
			e^{\alpha_h \tau_h + \alpha_c \tau_c} - 1
		}
	\end{align*}
\end{block}
\end{frame}

\begin{frame}{Endoodwracalny silnik Otto}
\begin{block}{Ciepła}
	\begin{align*}
		Q_{2-3} &= n c_V \pab{T_3 - T_2} =
			n c_V \frac{
				\pab{\kappa T_h - T_c}\pab{e^{\alpha_h \tau_h} - 1}\pab{e^{\alpha_c \tau_c} - 1}
			}{
				\kappa \pab{e^{\alpha_h \tau_h - \alpha_c \tau_c} - 1}
			}, \\
		Q_{4-1} &= n c_V \pab{T_1 - T_4} =
			- n c_V \frac{
				\pab{\kappa T_h - T_c}\pab{e^{\alpha_h \tau_h} - 1}\pab{e^{\alpha_c \tau_c} - 1}
			}{
				e^{\alpha_h \tau_h - \alpha_c \tau_c} - 1
			}
	\end{align*}
\end{block}
\begin{block}{Moc}
	\begin{multline*}
		P = \frac{n c_V}{\zeta\pab{\tau_h + \tau_c}}
		\frac{\pab{1 - \kappa}\pab{\kappa T_h - T_c}}{\kappa} \\
		\sinh\pab{\frac{\alpha_h \tau_h}{2}}
		\sinh\pab{\frac{\alpha_c \tau_c}{2}}
		\cosh\pab{\frac{\alpha_h \tau_h + \alpha_c \tau_c}{2}}
	\end{multline*}
\end{block}
\begin{block}{Sprawność}
	\begin{equation*}
		\eta = 1 - \kappa = 1 - \sqrt{\frac{T_c}{T_h}}
	\end{equation*}
\end{block}
\end{frame}

		\end{block}
